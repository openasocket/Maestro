import hashlib
import json
import subprocess
from pathlib import Path

out = Path('tmp/grok-tier1-audit/branches')
out.mkdir(parents=True, exist_ok=True)
roots = ['src/main', 'src/shared', 'src/renderer', 'src/web', 'src/cli', 'src/__tests__', 'docs']
roots += ['CLAUDE.md', 'CLAUDE-AGENTS.md', 'PROVIDER-SUPPORT.md', 'CONTRIBUTING.md', 'README.md']
patterns = {
    'providers': r'claude|codex|antigravity|grok',
    'registries': r'AGENT_IDS|PICKABLE_AGENT_IDS|AGENT_PICKER_META|SUPPORTED_AGENTS|AGENT_AUTOSELECT_ORDER|PROVIDER_PROFILE_DEFINITIONS|hasCapability|supports[A-Z]|getAgentCapabilities|createTranslationRegistry',
    'branches': r'(===|!==|case |includes\(|new Set|Record<|PROVIDERS|AGENTS|TYPES|ALLOWLIST|allowlist|whitelist|providerId|toolType).*(claude|codex|antigravity|grok)|(claude|codex|antigravity|grok).*(===|!==|includes\(|:)',
}
receipts = []
for name, pattern in patterns.items():
    command = ['rg', '--no-heading', '--color', 'never', '--sort', 'path', '-n', '-i', '-e', pattern, *roots]
    result = subprocess.run(command, capture_output=True)
    assert result.returncode == 0, result.stderr.decode()
    target = out / (name + '.txt')
    target.write_bytes(result.stdout)
    matches = result.stdout.decode().rstrip('\n').split('\n')
    by_root = {}
    for root in roots:
        lines = [line for line in matches if line.startswith(root + ('/' if Path(root).is_dir() else ':'))]
        by_root[root] = {'matchingLines': len(lines), 'files': len({line.split(':', 1)[0] for line in lines})}
    receipts.append({'query': name, 'command': command, 'exitCode': result.returncode,
                     'outputPath': str(target), 'sha256': hashlib.sha256(result.stdout).hexdigest(),
                     'matchingLines': len(matches), 'files': len({line.split(':', 1)[0] for line in matches}), 'byRoot': by_root})
(out / 'search-receipts.json').write_text(json.dumps(receipts, indent=2) + '\n')
print(json.dumps(receipts, indent=2))
